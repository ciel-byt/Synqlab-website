```python
def build_agent_artifact_path(
    artifact_root: Path,
    namespace: str,
    *sub_parts: str,
) -> Path:
    """Build canonical path under agent artifact root.

    Examples:
        build_agent_artifact_path(root, SESSIONS, session_id, "abc.pdb")
        → root / sessions / <session_id> / abc.pdb

        build_agent_artifact_path(root, FOLDING, "omegafold", "hash.pdb")
        → root / folding / omegafold / hash.pdb

        build_agent_artifact_path(root, PIPELINES, f"{run_id}.json")
        → root / pipelines / <run_id>.json
    """
    result = artifact_root / namespace
    for part in sub_parts:
        result = result / part
    return result
```

```python
from __future__ import annotations

class TestNamespaceConstants:

        assert PIPELINES == "pipelines"

    def test_folding(self):
        assert FOLDING == "folding"

    def test_candidates(self):
        assert CANDIDATES == "candidates"

    def test_foldx(self):
        assert FOLDX == "foldx"

    def test_sandbox(self):
        assert SANDBOX == "sandbox"

    def test_bundled(self):
        assert BUNDLED == "bundled"


class TestBuildAgentArtifactPath:
    """build_agent_artifact_path joins root + namespace + sub_parts."""

    @pytest.fixture()
    def root(self) -> Path:
        return Path("data/agent_artifacts")

    def test_namespace_only(self, root: Path):
        result = build_agent_artifact_path(root, SESSIONS)
        assert result == root / "sessions"

    def test_session_with_id(self, root: Path):
        result = build_agent_artifact_path(root, SESSIONS, "abc-123")
        assert result == root / "sessions" / "abc-123"

    def test_session_with_id_and_file(self, root: Path):
        result = build_agent_artifact_path(root, SESSIONS, "abc-123", "out.pdb")
        assert result == root / "sessions" / "abc-123" / "out.pdb"

    def test_folding_engine_scoped(self, root: Path):
        result = build_agent_artifact_path(root, FOLDING, "omegafold", "hash.pdb")
        assert result == root / "folding" / "omegafold" / "hash.pdb"

    def test_folding_esmfold(self, root: Path):
        result = build_agent_artifact_path(root, FOLDING, "esmfold", "hash.fasta")
        assert result == root / "folding" / "esmfold" / "hash.fasta"

    def test_candidates_run_scoped(self, root: Path):
        result = build_agent_artifact_path(root, CANDIDATES, "run-42", "hash.pdb")
        assert result == root / "candidates" / "run-42" / "hash.pdb"

        result = build_agent_artifact_path(root, PIPELINES, "run-99.json")

    def test_foldx_scratch(self, root: Path):

        assert result == root / "foldx" / "abc12345"

        result = build_agent_artifact_path(root, SANDBOX)
        assert result == root / "sandbox"
```
