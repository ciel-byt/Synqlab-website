# Plan 27-01: Backend Storage Browse & Download Endpoints

## Objective

Add two new FastAPI endpoints to `app/main.py` that enable the frontend Storage browser to list directory contents and download individual files from the `data/agent_artifacts/` namespace tree.

## must_haves

- [ ] `GET /api/storage/browse` endpoint returns JSON directory listing with name, is_dir, size, modified, path
- [ ] `GET /api/storage/download` endpoint returns file as `FileResponse` with `Content-Disposition: attachment`
- [ ] Path traversal attack prevented via `Path.resolve()` + prefix check
- [ ] Browsing root (`namespace=""`) lists all top-level namespaces
- [ ] Empty directories return `{"entries": [], ...}` (not 404)

```python
@app.get("/api/storage/browse")
async def browse_storage(namespace: str = "", path: str = ""):
    """Browse the agent artifact directory tree.

    Returns a directory listing for the given namespace/path combination.
    Root listing (no args) returns all top-level namespace folders.
    """
    import os
    from datetime import datetime, timezone

    root = Path(os.environ.get("SYNQLAB_ARTIFACT_DIR", "data/agent_artifacts"))
    target = (root / namespace / path) if namespace else root
    target = target.resolve()

    # Security: prevent path traversal
    if not str(target).startswith(str(root.resolve())):
        raise HTTPException(status_code=403, detail="Access denied")

    if not target.exists():
        return {"entries": [], "namespace": namespace, "path": path}

    if not target.is_dir():
        raise HTTPException(status_code=400, detail=f"Not a directory: {namespace}/{path}")

    entries = []
    try:
        for entry in sorted(target.iterdir()):
            # Skip hidden files (.gitkeep, etc.)
            if entry.name.startswith("."):
                continue
            try:
                stat = entry.stat()
                entries.append({
                    "name": entry.name,
                    "is_dir": entry.is_dir(),
                    "size": stat.st_size if entry.is_file() else None,
                    "modified": datetime.fromtimestamp(stat.st_mtime, tz=timezone.utc).isoformat(),
                    "path": str(entry.relative_to(root)).replace("\\", "/"),
                })
            except OSError:
                continue
    except PermissionError:
        raise HTTPException(status_code=403, detail="Permission denied")

    return {"entries": entries, "namespace": namespace, "path": path}
```

Import `HTTPException` from `fastapi` if not already imported (it likely is — check existing imports). Import `Path` from `pathlib` if not already at module level.

```python
@app.get("/api/storage/download")
async def download_storage_file(path: str):
    """Download a file from the agent artifact directory.

    Returns the file with Content-Disposition: attachment header.
    """
    import os
    from fastapi.responses import FileResponse

    root = Path(os.environ.get("SYNQLAB_ARTIFACT_DIR", "data/agent_artifacts"))
    target = (root / path).resolve()

    # Security: prevent path traversal
    if not str(target).startswith(str(root.resolve())):
        raise HTTPException(status_code=403, detail="Access denied")

    if not target.is_file():
        raise HTTPException(status_code=404, detail=f"File not found: {path}")

    # Determine MIME type from extension
    mime_map = {
        ".pdb": "chemical/x-pdb",
        ".fasta": "application/x-fasta",
        ".fa": "application/x-fasta",
        ".json": "application/json",
        ".csv": "text/csv",
        ".txt": "text/plain",
        ".pkl": "application/octet-stream",
        ".md": "text/markdown",
    }
    suffix = target.suffix.lower()
    media_type = mime_map.get(suffix, "application/octet-stream")

    return FileResponse(
        path=str(target),
        filename=target.name,
        media_type=media_type,
        headers={"Content-Disposition": f'attachment; filename="{target.name}"'},
    )
```
